\section{Fontes primárias, secundárias e inspeção própria}

\preencher{Classificar cada evidência: fonte primária (coletada pela equipe, que
esteve presente); fonte secundária (dado publicado por terceiros, citado com
autor, título e ano, por exemplo \citeonline{ticdomicilios}); inspeção própria
(navegação da equipe com NVDA e/ou apenas teclado, com a WCAG 2.2 em mãos ---
não constitui pesquisa com usuário).}

\section{Síntese dos resultados}

\preencher{Responder, com remissão às sessões (S1--S4) e às fontes: (a) quem usa,
com que frequência e em que circunstância; (b) de que recursos a pessoa dispõe;
(c) por que o fluxo importa e o que acontece quando não se conclui; (d) como a
tarefa é executada hoje e onde há hesitação, erro, pedido de ajuda ou desistência;
(e) o que a pessoa considera sucesso e quanto tempo e tentativas considera
aceitáveis; (f) diferenças entre perfis.}

\chapter{Contexto de uso}\label{cap:contexto}
% =====================================================================

Caracterização segundo os componentes da ISO 9241-11:2018 \cite{iso9241-11-2018}.
Em relação à NBR 9241-11 \cite{nbr9241-11}, equivalente à versão de 1998, a
edição de 2018 amplia o escopo para serviços e substitui ``equipamento'' pela
categoria mais ampla de ``recursos''.

\section{Usuários}
\preencher{Quem, frequência de uso, familiaridade digital, dependência de terceiros.}

\section{Metas e tarefas}
\preencher{O que a pessoa precisa obter e a consequência concreta da não conclusão.}

\section{Recursos}
\preencher{Dispositivo, conexão, franquia de dados, tempo disponível, documentos
exigidos, contas de acesso prévias e apoio humano disponível.}

\section{Ambiente}
\preencher{Condições técnicas, físicas, sociais e organizacionais do uso.}
